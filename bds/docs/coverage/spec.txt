# Strict event / state / hook spec, run on a real BDS by docs/coverage/run.mjs (→ docs/EVENTS.md).
# `## <id> | <condition>`: every section starts from the same reset (run.mjs RESET) and its own trigger;
# `~`/`!~` are regexes over everything printed since the section began (EV = event tap, ST = state tap,
# HOOK/CANCEL/BOOT = probe addon). `(?=.*k=v)` = field k has value v in any order. No two ids share a trigger.
# `## <id> | <condition> | 不可: <reason>` = nothing a player or script can do fires it (listed, not run).

# ---------- blocks: breaking ----------
## after.playerStartBreakingBlock | 掘り始めた瞬間。まだ壊れても止めてもいない（黒曜石を押し続けている）
setblock 2 -60 0 obsidian
@A dig 2 -60 0 hold
wait 500
~ ^EV after.playerStartBreakingBlock (?=.*block=minecraft:obsidian@2,-60,0)(?=.*player=A)(?=.*face=)
!~ ^EV after.playerCancelBreakingBlock
!~ ^EV (before|after).playerBreakBlock
@A release

## after.playerCancelBreakingBlock | 途中まで掘ったが壊し切らずにやめた（石を 5 tick だけ掘る）。ブロックは残る
setblock 2 -60 0 stone
@A dig 2 -60 0 5
wait 300
~ ^EV after.playerCancelBreakingBlock (?=.*block=minecraft:stone@2,-60,0)(?=.*breakProgress=0\.\d)(?=.*player=A)
!~ ^EV (before|after).playerBreakBlock
js dim.getBlock({x:2,y:-60,z:0}).typeId
= minecraft:stone

## after.playerBreakBlock | 最後まで掘って壊し切った（土）。壊れた後の block は air、壊れる前は brokenBlockPermutation
setblock 2 -60 0 dirt
@A dig 2 -60 0
~ ^EV after.playerBreakBlock (?=.*block=minecraft:air@2,-60,0)(?=.*brokenBlockPermutation=minecraft:dirt)(?=.*player=A)
!~ ^EV after.playerCancelBreakingBlock

## before.playerBreakBlock | 壊れる直前に呼ばれ、cancel で壊させない（金ブロックは probe が止める → 残る）
setblock 2 -60 0 gold_block
@A dig 2 -60 0
~ ^EV before.playerBreakBlock (?=.*block=minecraft:gold_block@2,-60,0)(?=.*player=A)
~ ^CANCEL before.playerBreakBlock gold_block
!~ ^EV after.playerBreakBlock
js dim.getBlock({x:2,y:-60,z:0}).typeId
= minecraft:gold_block

## after.entityHitBlock | ブロックを叩いた（壊せない岩盤を叩く）。掘れないので開始・破壊イベントの有無と無関係
setblock 2 -60 0 bedrock
@A dig 2 -60 0 hold
wait 300
~ ^EV after.entityHitBlock (?=.*damagingEntity=A)(?=.*hitBlock=minecraft:bedrock@2,-60,0)(?=.*blockFace=)
!~ ^EV (before|after).playerBreakBlock
@A release

## after.explosion | 爆発が起きた（水中の TNT: 爆発はするがブロックは壊れない → blockExplode なし）
effect A resistance 30 255 true
fill 5 -62 5 7 -59 7 water
setblock 6 -62 6 tnt
@A cmd give @s flint_and_steel
@A goto 3.5 3.5
@A useon 6 -62 6 1

~ ^EV after.explosion (?=.*source=minecraft:tnt)(?=.*dimension=overworld)
!~ ^EV after.blockExplode
fill 5 -62 5 7 -59 7 air

## after.blockExplode | 爆発でブロックが 1 つ壊れるたびに（陸の TNT → 周りの草・土）
effect A resistance 30 255 true
setblock 4 -60 -4 tnt
@A cmd give @s flint_and_steel
@A goto 2.5 -2.5
@A useon 4 -60 -4
@A goto -6.5 5.5
wait 4500
~ ^EV after.blockExplode (?=.*explodedBlockPermutation=minecraft:(grass_block|dirt))(?=.*source=minecraft:tnt)

## before.explosion | 爆発の直前に呼ばれ、cancel で爆発させない（x>20 の TNT は probe が止める → 周りが残る）
effect A resistance 30 255 true
tp A 22.5 -60 0.5
setblock 24 -60 0 tnt
@A cmd give @s flint_and_steel
@A useon 24 -60 0
@A goto 22.5 5.5
wait 4500
~ ^EV before.explosion (?=.*source=minecraft:tnt)
~ ^CANCEL before.explosion x>20
!~ ^EV after.(explosion|blockExplode)
js dim.getBlock({x:24,y:-61,z:0}).typeId
= minecraft:grass_block
